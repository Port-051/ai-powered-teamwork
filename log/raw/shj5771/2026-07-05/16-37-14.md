def solution(k, tangerine):
    fruit_dict = {}
    for i in tangerine:
        if i in fruit_dict:
            fruit_dict[i] += 1
        else:
            fruit_dict[i] = 1
            
    fruit_sort = sorted(fruit_dict.values(),reverse=True)
    result = 0
    
    for j in fruit_sort:
        if k-j >=0:
            k -= j
            result += 1
        else:
            continue
    if result == 0:
        result += 1
